cmake_minimum_required(VERSION 3.22)
project(taskbar_app_sync VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(BUILD_SHARED_LIBS ON)
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib")
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
find_package(ECM 6.0 REQUIRED NO_MODULE)
list(APPEND CMAKE_MODULE_PATH ${ECM_MODULE_PATH})
include(KDEInstallDirs6)
find_package(Qt6 6.6 REQUIRED COMPONENTS Core Gui Qml Quick DBus)
find_package(KF6CoreAddons REQUIRED)
find_package(KF6Config REQUIRED)
find_package(Plasma REQUIRED)
find_package(PlasmaQuick REQUIRED)
add_library(syncengine STATIC src/syncengine.cpp)
target_link_libraries(syncengine PUBLIC Qt6::Core Qt6::Qml)
target_include_directories(syncengine PUBLIC src)
kcoreaddons_add_plugin(taskbar_app_sync SOURCES src/companion.cpp INSTALL_NAMESPACE plasma/applets)
target_link_libraries(taskbar_app_sync PRIVATE syncengine Qt6::DBus Plasma::Plasma Plasma::PlasmaQuick KF6::ConfigCore KF6::ConfigQml)
plasma_install_package(package org.local.taskbar.sync)
set_target_properties(taskbar_app_sync PROPERTIES PREFIX "" OUTPUT_NAME org.local.taskbar.sync)
